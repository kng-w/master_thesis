\section{GNN によるレイアウト推定モデルの詳細}
\label{app:script2name_gnn}

\begin{table}[htbp]
  \centering
  \small
  \begin{threeparttable}
    \caption{Script2Direction2Layout における GNN（HeteroConv）モデル構成（実装に基づく）}
    \label{tab:gnn_arch}
    \begin{tabular}{p{0.33\linewidth} p{0.60\linewidth}}
      \toprule
      \textbf{項目} & \textbf{設定}                                           \\
      \midrule
      入力グラフ       & 見開き単位の異種グラフ（Heterogeneous graph）                      \\
      ノード種別       &
      Spread / Panel / Character / Character Instance / Speech            \\
      エッジ種別（例）    &
      action edges（Character Instance 間），
      Panel--Character Instance（包含），
      Speech の読順関係（順序）など                                                  \\
      ノード特徴       &
      テキスト特徴（Tokenizer による入力），および列挙型の演出属性（pose/action/emotion 等）\tnote{a} \\
      テキストエンコーダ   &
      scratch Transformer または pretrained HF モデル（QLoRA 対応）\tnote{b}        \\
      GNN 本体      &
      HeteroConv によるメッセージパッシング，層数 = \texttt{num\_layers}（既定3）             \\
      出力ノード       &
      Panel / Character Instance / Speech                                 \\
      出力形式        &
      混合密度ネットワーク（MDN）により $\mathbf{b}=(x,y,w,h)$ を推定                       \\
      混合数         &
      ガウス混合数 $K$ = \texttt{num\_gaussians}（既定5）                           \\
      損失関数        &
      MDN の負の対数尤度（NLL）                                                    \\
      \bottomrule
    \end{tabular}
    \begin{tablenotes}[flushleft]
      \footnotesize
      \item[a] 実装では node type ごとに列挙特徴（例：Character Instance の pose/action/emotion 等）を付与する（\texttt{extract\_enum\_features}）。
      \item[b] 事前学習テキストエンコーダ使用時は \texttt{text\_encoder\_freeze} と \texttt{text\_encoder\_qlora} により凍結・QLoRA を切替可能。
    \end{tablenotes}
  \end{threeparttable}
\end{table}
